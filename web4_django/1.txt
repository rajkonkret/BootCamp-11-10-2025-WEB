pip install Django==5.2.5 - instalacja wskazanej wersji Django
python -m django --version - sprawdzenie wersji
django-admin --version

django-admin startproject myproject - tworzenie projektu django

cd myproject
python manage.py runserver - uruchomienie projektu djangowego
gunicorn, uwsgi, uvicorn
gunicorn -b :5000 main:app - uruchommienie na porcie 5000
